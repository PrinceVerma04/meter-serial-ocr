# meterocr PTQ INT8 — Python 3.8 pins.  *** INSTALL INTO A VENV ***
#
#     python3.8 -m venv venv
#     ./venv/bin/pip install -r requirements-py38.txt
#
# Do NOT `pip3 install` these system-wide on the flaskML server: numpy 1.24.4
# will replace the numpy 1.19.5 that TensorFlow 2.4.1 needs and break it.
# See DEPLOY.md section 0 for the recovery steps if that already happened.
# ---------------------------------------------------------------------------

# CPU inference runtime.
#   * 1.13.1 is the FLOOR - older versions lack the com.microsoft:QLinearSoftmax
#     operator this model uses (1.12.0 fails to load outright, verified).
#   * 1.16.3 is the CEILING your server's pip can see: onnxruntime 1.17+ ships
#     manylinux_2_27/2_28 wheels whose tags old pip does not understand.
#     Upgrading pip would expose newer versions, but 1.16.3 is tested and fine.
onnxruntime==1.16.3

# Forced by BOTH constraints:
#   * onnxruntime >= 1.13 requires numpy >= 1.21.6
#   * numpy 1.25+ requires Python >= 3.9
# 1.24.4 is the only window that satisfies both. It is NOT compatible with
# TensorFlow 2.4.1 (which pins numpy~=1.19.2) - hence the venv.
numpy==1.24.4

# cp37-abi3 wheel, requires_python >=3.6 -> installs on 3.8.
# 'headless' avoids needing libGL on a server.
opencv-python-headless==4.9.0.80

# --- HTTP API only. Skip if you use batch_predict.py / import MeterOCR. ---
Flask==3.0.3
Werkzeug==3.0.6
Flask-Cors==5.0.0
gevent==24.2.1
